function esc(str) {
  return String(str ?? '').replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;').replace(/"/g,'&quot;');
}

function getAccents(tmpl) {
  return ACCENTS[tmpl] ?? ACCENTS.riot;
}

// Simple debounce – returns a fn that delays `fn` by `ms`
function debounce(fn, ms) {
  let timer;
  return function(...args) {
    clearTimeout(timer);
    timer = setTimeout(() => fn.apply(this, args), ms);
  };
}

/* ═══════════════════════════════════════
   RENDER: ACCENT DOTS
═══════════════════════════════════════ */
function renderAccents() {
  const row = document.getElementById('accent-row');
  row.innerHTML = '';
  getAccents(state.template).forEach(ac => {
    const d = document.createElement('div');
    // FIX: compare against state.accent, not position index
    d.className = 'accent-dot' + (ac === state.accent ? ' active' : '');
    d.style.background = ac;
    d.dataset.ac = ac;
    d.addEventListener('click', () => setAccent(ac));
    row.appendChild(d);
  });
}

/* ═══════════════════════════════════════
   ACTIONS
═══════════════════════════════════════ */

  state.template = id;
  state.accent   = getAccents(id)[0];
  document.getElementById('app').style.setProperty('--ac', state.accent);
  document.querySelectorAll('.tmpl-btn').forEach(b =>
    b.classList.toggle('active', b.dataset.id === id)
  );
  renderAccents();
  renderTemplate();
}

function setAccent(ac) {

function showPhotoError(msg) {
  const el = document.getElementById('photo-error');
  el.textContent = msg;
  el.style.display = msg ? 'block' : 'none';
}

function handlePhotoChange(e) {
  const file = e.target.files[0];
  if (!file) return;

  // Validate type
  if (!file.type.startsWith('image/')) {
    showPhotoError('File harus berupa gambar.');
    e.target.value = '';
    return;
  }
  // Validate size
  if (file.size > MAX_PHOTO_BYTES) {
    showPhotoError('Ukuran foto maks 8 MB.');
    e.target.value = '';
    return;
  }
  showPhotoError('');

  const reader = new FileReader();
  reader.onerror = () => showPhotoError('Gagal membaca file.');
  reader.onload = evt => {
    state.photo = evt.target.result;
    const thumb = document.getElementById('photo-thumb');
    thumb.src   = state.photo;
    thumb.style.display = 'block';
    document.getElementById('photo-remove').style.display = 'block';
    const short = file.name.length > 18
      ? file.name.substring(0, 16) + '…'
      : file.name;
    document.getElementById('photo-label-btn').textContent = '✓ ' + short;
    renderTemplate();
  };
  reader.readAsDataURL(file);
}

/* ═══════════════════════════════════════
   EXPORT
═══════════════════════════════════════ */
async function exportZine() {
  const btn = document.getElementById('export-btn');
  btn.disabled = true;
  btn.textContent = '⏳ RENDERING...';

  try {
    // Ensure fonts are loaded
    await document.fonts.ready;

    const container = document.getElementById('export-canvas-container');
    container.innerHTML = buildTemplate(state, EXPORT_SCALE);
    const el = container.firstElementChild;

    // Wait for all images inside the export element to finish loading
    const imgs = Array.from(el.querySelectorAll('img'));
    await Promise.all(imgs.map(img =>
      img.complete
        ? Promise.resolve()
        : new Promise(resolve => { img.onload = resolve; img.onerror = resolve; })
    ));

    // Small buffer to let browser repaint
    await new Promise(r => setTimeout(r, 400));

    const canvas = await html2canvas(el, {
      scale:           1,
      useCORS:         true,
      allowTaint:      true,
      backgroundColor: null,
      logging:         false,
      width:           1080,
      height:          1920,
      imageTimeout:    10000,
    });

    // Trigger download
    const link  = document.createElement('a');
    const stamp = new Date().toISOString().slice(0,10);
    link.download = `berzine-${state.template.toUpperCase()}-${stamp}.png`;
    link.href = canvas.toDataURL('image/png');
    link.click();

    btn.textContent = '✓ TERSIMPAN!';
    setTimeout(() => {
      btn.textContent = '↓ SIMPAN SEBAGAI PNG';
      btn.disabled = false;
    }, 2500);

  } catch (err) {
    console.error('[BERZINE] Export error:', err);
    btn.textContent = '↓ SIMPAN SEBAGAI PNG';
    btn.disabled = false;
    alert('Gagal export. Coba lagi ya!');
  }
}

/* ═══════════════════════════════════════
   BUILD TEMPLATE GRID
═══════════════════════════════════════ */
function buildTmplGrid() {
  const grid = document.getElementById('tmpl-grid');
  TEMPLATES.forEach((t, i) => {
    const btn = document.createElement('button');
    btn.type      = 'button';
    btn.className = 'tmpl-btn' + (i === 0 ? ' active' : '');
    btn.dataset.id = t.id;
    btn.innerHTML = `${t.name}<span class="tmpl-desc">${t.desc}</span>`;
    btn.addEventListener('click', () => setTemplate(t.id));
    grid.appendChild(btn);
  });
}
